Sincronização com a Planilha Google Sheets

Sincronização bidirecional automática a cada 5s (leitura e envio) na planilha oficial do casamento Laís & Hans

1 Auto-Sync Bidirecional a cada 5 Segundos (Leitura + Envio Automático!)

Esta aplicação sincroniza automaticamente com a planilha a cada 5 segundos em segundo plano — buscando novos nomes/edições da planilha e também reenviando automaticamente qualquer alteração pendente para a planilha. Assim, se a Laís e o Hans estiverem usando ao mesmo tempo em aparelhos diferentes, a lista fica sempre idêntica sem faltar nenhum nome.

1. Abra a planilha oficial no Google Sheets e clique em Extensões > Apps Script.

2. Apague o código antigo, cole o código abaixo e clique no ícone de disquete (Salvar).

3. Clique em Implantar > Nova implantação (ou Gerenciar implantações > Editar > Nova versão), escolha App da Web, selecione Executar como: Eu e Quem pode acessar: Qualquer pessoa, e copie a URL terminada em /exec.

1. No navegador Chrome do celular, abra o link da planilha, toque nos 3 pontinhos (⋮) e marque "Para computador" (Versão para desktop).

3. Toque em Implantar > Nova implantação, configure como App da Web aberto para Qualquer pessoa e copie o link /exec.

Código Oficial Google Apps Script (v4.2.0 — Sem Conflito de const): 
var SHEET_ID = '1_w4UCIG0wmtXetsUxItOqN9zbno7JE7IWcXPGwu4oag';

var OFFICIAL_HEADER = ['Nome', 'Sobrenome', 'Relação com os Noivos', 'Especificação (Outros) / Padrinho ou Madrinha', 'Celular / WhatsApp', 'Instagram', 'Tem Acompanhante', 'Data de Inclusão', 'Cidade', 'Padrinho / Madrinha'];

function normStr(s) {
  return String(s || '').trim().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/\s+/g, ' ');
}

function getTargetSheet() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheet = ss.getSheets()[0];
  const lastRow = sheet.getLastRow();
  if (lastRow === 0) {
    sheet.appendRow(OFFICIAL_HEADER);
  } else {
    const firstCell = normStr(sheet.getRange(1, 1).getValue());
    if (firstCell !== 'nome') {
      sheet.insertRowBefore(1);
      sheet.getRange(1, 1, 1, 10).setValues([OFFICIAL_HEADER]);
    }
  }
  return sheet;
}

function compactAndOrderSheet(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow === 0) {
    sheet.appendRow(OFFICIAL_HEADER);
    return;
  }
  const allValues = sheet.getRange(1, 1, lastRow, 10).getValues();
  const validRows = [];

  for (let i = 0; i < allValues.length; i++) {
    const r = allValues[i];
    const nome = String(r[0] || '').trim();
    const sobrenome = String(r[1] || '').trim();
    if (!nome && !sobrenome) continue;
    if (normStr(nome) === 'nome' && (normStr(sobrenome) === 'sobrenome' || i === 0)) continue;
    if (nome.indexOf('__SLOT_') === 0) continue;
    validRows.push(r);
  }

  sheet.clearContents();
  const finalMatrix = [OFFICIAL_HEADER].concat(validRows);
  sheet.getRange(1, 1, finalMatrix.length, 10).setValues(finalMatrix);
}

function doGet(e) {
  const lock = LockService.getScriptLock();
  try { lock.waitLock(10000); } catch (err) {}
  try {
    const sheet = getTargetSheet();
    const p = (e && e.parameter) ? e.parameter : {};
    const action = p.action || '';

    if (action === 'add') {
      sheet.appendRow([
        p.nome || '',
        p.sobrenome || '',
        p.relacao || '',
        p.outros || '',
        p.whatsapp || '',
        p.instagram || '',
        p.acompanhante || 'Não',
        new Date().toLocaleString('pt-BR'),
        p.cidade || '',
        p.padrinho || p.outraCidade || ''
      ]);
      compactAndOrderSheet(sheet);
    } else if (action === 'delete') {
      const allRows = sheet.getDataRange().getValues();
      const origNome = normStr(p.origNome || p.nome || '');
      const origSobrenome = normStr(p.origSobrenome || p.sobrenome || '');
      for (let r = allRows.length - 1; r >= 1; r--) {
        if (normStr(allRows[r][0]) === origNome && normStr(allRows[r][1]) === origSobrenome) {
          sheet.deleteRow(r + 1);
          break;
        }
      }
      compactAndOrderSheet(sheet);
    } else if (action === 'edit') {
      const allRows = sheet.getDataRange().getValues();
      const origNome = normStr(p.origNome || '');
      const origSobrenome = normStr(p.origSobrenome || '');
      let matched = false;
      for (let r = 1; r < allRows.length; r++) {
        if (normStr(allRows[r][0]) === origNome && normStr(allRows[r][1]) === origSobrenome) {
          sheet.getRange(r + 1, 1, 1, 10).setValues([[
            p.nome || '',
            p.sobrenome || '',
            p.relacao || '',
            p.outros || '',
            p.whatsapp || '',
            p.instagram || '',
            p.acompanhante || '',
            new Date().toLocaleString('pt-BR'),
            p.cidade || '',
            p.padrinho || p.outraCidade || ''
          ]]);
          matched = true;
          break;
        }
      }
      if (!matched && origNome && origNome.indexOf('__slot_') !== 0) {
        for (let r = 1; r < allRows.length; r++) {
          if (normStr(allRows[r][0]) === origNome) {
            sheet.getRange(r + 1, 1, 1, 10).setValues([[
              p.nome || '',
              p.sobrenome || '',
              p.relacao || '',
              p.outros || '',
              p.whatsapp || '',
              p.instagram || '',
              p.acompanhante || '',
              new Date().toLocaleString('pt-BR'),
              p.cidade || '',
              p.padrinho || p.outraCidade || ''
            ]]);
            break;
          }
        }
      }
      if (!p.skipCompact) {
        compactAndOrderSheet(sheet);
      }
    }

    const rows = sheet.getDataRange().getValues();
    const payload = JSON.stringify({ ok: true, rows: rows });
    if (p.callback) {
      return ContentService.createTextOutput(p.callback + '(' + payload + ');')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(payload)
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    try { lock.releaseLock(); } catch (err) {}
  }
}
